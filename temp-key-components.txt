import useCountdown from '../../hooks/useCountdown'

export default function CountdownTimer({ startAt, eventStatus }) {
  const remaining = useCountdown(eventStatus === 'ended' || eventStatus === 'cancelled' || eventStatus === 'live' ? null : startAt)
  if (eventStatus === 'cancelled') return <p className="pricing-countdown-status">The Masterclass Has Been Cancelled</p>
  if (eventStatus === 'ended') return <p className="pricing-countdown-status">The Masterclass Has Ended</p>
  if (eventStatus === 'live' || remaining === 0) {
    return <p className="pricing-countdown-status">The Masterclass Is Now Live</p>
  }
  if (remaining === null) {
    return <p className="pricing-countdown-status">Webinar Date To Be Announced</p>
  }

  const units = [
    ['Days', Math.floor(remaining / 86400)],
    ['Hours', Math.floor(remaining / 3600) % 24],
    ['Minutes', Math.floor(remaining / 60) % 60],
    ['Seconds', remaining % 60],
  ]

  return (
    <div className="pricing-countdown" role="timer" aria-label="Time until the masterclass starts">
      <p className="pricing-label">Masterclass Starts In</p>
      <dl>
        {units.map(([label, value]) => (
          <div key={label}>
            <dt>{label}</dt>
            <dd>{String(value).padStart(2, '0')}</dd>
          </div>
        ))}
      </dl>
    </div>
  )
}

import { motion, useReducedMotion } from 'framer-motion'

import { useId, useRef, useState } from 'react'

export default function FAQAccordion({ items, entrance }) {
  const id = useId()
  const buttons = useRef([])
  const [openIndex, setOpenIndex] = useState(0)
  const reducedMotion = useReducedMotion()

  function moveFocus(event, index) {
    const last = items.length - 1
    const positions = {
      ArrowDown: index === last ? 0 : index + 1,
      ArrowUp: index === 0 ? last : index - 1,
      Home: 0,
      End: last,
    }
    if (!Object.hasOwn(positions, event.key)) return

    event.preventDefault()
    buttons.current[positions[event.key]]?.focus()
  }

  return (
    <div className="faq-accordion">
      {items.map(({ question, answer }, index) => {
        const expanded = openIndex === index
        const triggerId = `${id}-question-${index}`
        const panelId = `${id}-answer-${index}`
        const Icon = expanded ? Minus : Plus

        return (
          <motion.div className="faq-item" key={question} data-open={expanded} {...entrance((index % 4) * 0.045, 6)}>
            <h3>
              <button
                id={triggerId}
                type="button"
                ref={(button) => { buttons.current[index] = button }}
                aria-expanded={expanded}
                aria-controls={panelId}
                onClick={() => setOpenIndex(expanded ? null : index)}
                onKeyDown={(event) => moveFocus(event, index)}
              >
                {question}<Icon size={21} aria-hidden="true" strokeWidth={1.8} />
              </button>
            </h3>
            <motion.div
              id={panelId}
              role="region"
              aria-labelledby={triggerId}
              inert={!expanded}
              className="faq-answer"
              initial={false}
              animate={{ height: expanded ? 'auto' : 0, opacity: expanded ? 1 : 0 }}
              transition={{ duration: reducedMotion ? 0 : 0.24, ease: 'easeInOut' }}
            >
              <p>{answer}</p>
            </motion.div>
          </motion.div>
        )
      })}
    </div>
  )
}

import { animate, useInView, useReducedMotion } from 'framer-motion'
import { useEffect, useRef, useState } from 'react'

const numberFormat = new Intl.NumberFormat('en-IN')

export default function AnimatedCounter({ value, suffix = '' }) {
  const counter = useRef(null)
  const inView = useInView(counter, { once: true })
  const reducedMotion = useReducedMotion()
  const [count, setCount] = useState(0)

  useEffect(() => {
    if (!inView || reducedMotion) return

    const animation = animate(0, value, {
      duration: 1.5,
      ease: 'easeOut',
      onUpdate: (current) => setCount(Math.round(current)),
    })

    return () => animation.stop()
  }, [inView, reducedMotion, value])

  return (
    <span ref={counter}>
      <span aria-hidden="true">{numberFormat.format(reducedMotion ? value : count)}{suffix}</span>
      <span className="sr-only">{numberFormat.format(value)}{suffix}</span>
    </span>
  )
}

function VideoCarousel({ onPlay, reducedMotion, entrance }) {
  const track = useRef(null)
  const [positions, setPositions] = useState([0])
  const [active, setActive] = useState(0)

  useEffect(() => {
    const element = track.current
    const observer = new ResizeObserver(() => {
      const limit = element.scrollWidth - element.clientWidth
      const start = element.firstElementChild.offsetLeft
      const offsets = Array.from(element.children, (child) => Math.min(limit, child.offsetLeft - start))
      setPositions([...new Set(offsets)])
    })
    observer.observe(element)
    return () => observer.disconnect()
  }, [])

  function scrollTo(index) {
    const left = positions[index]
    if (left === undefined) return

    track.current.scrollTo({ left, behavior: reducedMotion ? 'instant' : 'smooth' })
  }

  function updateActive() {
    const left = track.current.scrollLeft
    setActive(positions.reduce((closest, position, index) =>
      Math.abs(position - left) < Math.abs(positions[closest] - left) ? index : closest, 0))
  }

  useEffect(updateActive, [positions])

  function onKeyDown(event) {
    if (event.target !== track.current && !event.target.matches('.testimonial-play')) return
    if (event.key !== 'ArrowLeft' && event.key !== 'ArrowRight') return

    event.preventDefault()
    scrollTo(active + (event.key === 'ArrowRight' ? 1 : -1))
  }

  return (
    <div className="testimonial-carousel" role="region" aria-label="Video testimonials" aria-roledescription="carousel">
      <ul
        id="testimonial-video-track"
        className="testimonial-track"
        ref={track}
        tabIndex={positions.length > 1 ? 0 : undefined}
        onKeyDown={onKeyDown}
        onScroll={updateActive}
      >
        {videoEntries.map((entry, index) => (
          <motion.li key={entry.id} {...entrance((index % 3) * 0.07)}>
            <TestimonialVideoCard entry={entry} onPlay={onPlay} />
          </motion.li>
        ))}
      </ul>
      {positions.length > 1 && (
        <div className="testimonial-navigation">
          <button type="button" aria-label="Previous testimonials" aria-controls="testimonial-video-track" disabled={active === 0} onClick={() => scrollTo(active - 1)}>
            <ChevronLeft size={19} aria-hidden="true" />
          </button>
          {positions.map((position, index) => (
            <button
              key={position}
              type="button"
              className="testimonial-page-dot"
              aria-label={`Go to testimonial position ${index + 1}`}
              aria-current={active === index ? 'true' : undefined}
              aria-controls="testimonial-video-track"
              onClick={() => scrollTo(index)}
            />
          ))}
          <button type="button" aria-label="Next testimonials" aria-controls="testimonial-video-track" disabled={active === positions.length - 1} onClick={() => scrollTo(active + 1)}>
            <ChevronRight size={19} aria-hidden="true" />
          </button>
        </div>
      )}
    </div>
  )
}
